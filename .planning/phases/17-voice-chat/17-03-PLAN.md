## Task 1: Add Voice State to ChatView

<action>
In `ChatView.swift`, add these state properties:

```swift
@State private var speechService = SpeechRecognitionService()
@State private var showVoiceSheet = false
@State private var hasMicPermission: Bool? = nil  // nil = not yet requested
```

Set up the `onSilenceTimeout` callback in `.onAppear`:
```swift
.onAppear {
    speechService.onSilenceTimeout = {
        // Auto-stop after 2s silence, fill transcript
        speechService.stopListening()
        if !speechService.transcript.isEmpty {
            inputText = speechService.transcript
        }
        showVoiceSheet = false
        
        let generator = UINotificationFeedbackGenerator()
        generator.notificationOccurred(.success)
    }
}
```

## Task 2: Replace Send Button with Mic/Send Toggle

<action>
Replace the current Send button (lines 65-74 in ChatView.swift) with a dynamic toggle:

```swift
// Dynamic Mic / Send button
Button(action: {
    if inputText.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty {
        // MIC MODE: Start voice input
        Task {
            await handleMicTap()
        }
    } else {
        // SEND MODE: Send message
        sendMessage()
    }
}) {
    Image(systemName: inputText.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty ? "mic.fill" : "arrow.up.circle.fill")
        .resizable()
        .frame(width: 32, height: 32)
        .foregroundColor(inputText.trimmingCharacters(in: .whitespacesAndNewlines).isEmpty ? .green : .green)
        .contentTransition(.symbolEffect(.replace))
}
.disabled(viewModel.isStreaming)
.padding(.bottom, 4)
```

Add the `handleMicTap()` function:
```swift
private func handleMicTap() async {
    // First-time permission request (4A - Best Practice)
    if hasMicPermission == nil {
        let granted = await speechService.requestAuthorization()
        hasMicPermission = granted
        
        // Also request mic permission
        let micGranted = await AVAudioApplication.requestRecordPermission()
        hasMicPermission = granted && micGranted
    }
    
    guard hasMicPermission == true else {
        viewModel.errorMessage = "Vui lòng cấp quyền micro và nhận diện giọng nói trong Cài đặt."
        return
    }
    
    // Start listening
    let generator = UIImpactFeedbackGenerator(style: .light)
    generator.impactOccurred()
    
    speechService.startListening()
    showVoiceSheet = true
}
```

Add `import AVFoundation` at the top of the file.
</action>

## Task 3: Add Voice Recording Sheet Overlay

<action>
Add the VoiceRecordingSheet as an overlay at the bottom of the ChatView, just before `.navigationTitle`:

```swift
.overlay(alignment: .bottom) {
    if showVoiceSheet {
        VoiceRecordingSheet(
            speechService: speechService,
            onDismiss: {
                speechService.stopListening()
                // Fill transcript into input if available
                if !speechService.transcript.isEmpty {
                    inputText = speechService.transcript
                }
                showVoiceSheet = false
            },
            onConfirm: { text in
                inputText = text
                showVoiceSheet = false
            }
        )
        .transition(.move(edge: .bottom).combined(with: .opacity))
        .animation(.spring(response: 0.3, dampingFraction: 0.8), value: showVoiceSheet)
        .padding(.bottom, 80)  // Above the input area
    }
}
```
</action>

</acceptance_criteria>
