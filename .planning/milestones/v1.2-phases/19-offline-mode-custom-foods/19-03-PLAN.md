3. **Local foods**: Existing `searchLocalFoods()` minus custom and recent duplicates

Add management methods:
```swift
func deleteCustomFood(_ food: FoodItemModel) async {
    deletedFood = food
    try? await foodRepository.deleteFood(by: food.id)
    showUndoToast = true
    // Remove from customResults immediately
    customResults.removeAll { $0.id == food.id }
    HapticManager.light()
}

func undoDelete() async {
    if let food = deletedFood {
        try? await foodRepository.saveCustomFood(food)
        deletedFood = nil
        showUndoToast = false
        // Re-search to refresh
        await performSearch(query: searchText)
    }
}

func duplicateFood(_ food: FoodItemModel) async {
    let duplicate = try? await foodRepository.duplicateCustomFood(food)
    if duplicate != nil {
        HapticManager.success()
        await performSearch(query: searchText)
    }
}
```

var body: some View {
    ZStack(alignment: .bottom) {
        List {
            if viewModel.searchText.isEmpty {
                // When idle: show custom foods + suggestions
                if !viewModel.customResults.isEmpty {
                    customFoodsSection
                }
                if !viewModel.suggestions.isEmpty {
                    Section(header: Text("Gợi ý")) { /* existing */ }
                }
            } else {
                // ⭐ Section 1: Custom foods (D-25, D-27 — only if has data)
                if !viewModel.customResults.isEmpty {
                    customFoodsSection
                }
                // 🕘 Section 2: Recent (D-26)
                if !viewModel.recentResults.isEmpty {
                    Section(header: recentHeader) {
                        ForEach(viewModel.recentResults) { food in foodRow(for: food) }
                    }
                }
                // 📦 Section 3: Local offline
                if !viewModel.localResults.isEmpty {
                    Section(header: Text("📦 Dữ liệu offline")) {
                        ForEach(viewModel.localResults) { food in foodRow(for: food) }
                    }
                }
                // Empty state with CTA (D-12)
                if viewModel.customResults.isEmpty && viewModel.localResults.isEmpty && 
                   viewModel.recentResults.isEmpty && !viewModel.isSearchingAPI && viewModel.apiResults.isEmpty {
                    emptyStateWithCTA
                }
                // 🌐 Section 4: API
                if viewModel.isSearchingAPI { /* loading */ }
                if !viewModel.apiResults.isEmpty {
                    Section(header: Text("🌐 CalorieNinjas")) {
                        ForEach(viewModel.apiResults) { food in foodRow(for: food) }
                    }
                }
            }
        }
        
        // Undo toast (D-22)
        if viewModel.showUndoToast {
            undoToastView
        }
    }
    .toolbar {

        CustomFoodBuilderSheet(/* callbacks */)
    }

        CustomFoodBuilderSheet(existingFood: food, /* callbacks */)
    }
