## Task 1: Create PendingChatQueue service

Create `LiiO_EatClean/Features/Chat/PendingChatQueue.swift`:

```swift
import Foundation
import CoreData

struct PendingMessage: Identifiable, Equatable {
    let id: UUID
    let text: String
    let createdAt: Date
    let conversationID: UUID
    var status: PendingStatus
    
    enum PendingStatus: String {
        case pending = "pending"
        case sending = "sending"
        case failed = "failed"
    }
}

@Observable
class PendingChatQueue {
    static let shared = PendingChatQueue()
    
    var pendingMessages: [PendingMessage] = []
    
    private let context: NSManagedObjectContext
    
    private init() {
        self.context = PersistenceController.shared.container.viewContext
        loadPendingMessages()
        observeConnectivity()
    }
    
    // MARK: - Persistence (D-33)
    
    private func loadPendingMessages() {
        let request: NSFetchRequest<PendingChatMessage> = PendingChatMessage.fetchRequest()
        request.sortDescriptors = [NSSortDescriptor(keyPath: \PendingChatMessage.createdAt, ascending: true)]
        
        if let entities = try? context.fetch(request) {
            pendingMessages = entities.map { entity in
                PendingMessage(
                    id: entity.id ?? UUID(),
                    text: entity.text ?? "",
                    createdAt: entity.createdAt ?? Date(),
                    conversationID: entity.conversationID ?? UUID(),
                    status: PendingMessage.PendingStatus(rawValue: entity.status ?? "pending") ?? .pending
                )
            }
        }
    }
    
    // MARK: - Queue Management (D-31, D-32)
    
    func enqueue(text: String, conversationID: UUID) {
        let message = PendingMessage(
            id: UUID(),
            text: text,
            createdAt: Date(),
            conversationID: conversationID,
            status: .pending
        )
        pendingMessages.append(message)
        
        // Persist to CoreData
        let entity = PendingChatMessage(context: context)
        entity.id = message.id
        entity.text = message.text
        entity.createdAt = message.createdAt
        entity.conversationID = message.conversationID
        entity.status = "pending"
        try? context.save()
    }
    
    func markSending(id: UUID) {
        if let index = pendingMessages.firstIndex(where: { $0.id == id }) {
            pendingMessages[index].status = .sending
            updateEntity(id: id, status: "sending")
        }
    }
    
    func markFailed(id: UUID) {
        if let index = pendingMessages.firstIndex(where: { $0.id == id }) {
            pendingMessages[index].status = .failed
            updateEntity(id: id, status: "failed")
        }
    }
    
    func remove(id: UUID) {
        pendingMessages.removeAll { $0.id == id }
        deleteEntity(id: id)
    }
    
    // MARK: - Auto-Retry on Reconnect (D-10)
    
    private func observeConnectivity() {
        // Observe NetworkMonitor.shared.isConnected changes
        // When transitions from false → true, call retryPending()
        Task { @MainActor in
            var wasOffline = !NetworkMonitor.shared.isConnected
            // Poll-based observation (simple approach)
            while true {
                try? await Task.sleep(for: .seconds(2))
                let isNowOnline = NetworkMonitor.shared.isConnected
                if wasOffline && isNowOnline {
                    await retryPending()
                }
                wasOffline = !isNowOnline
            }
        }
    }
    
    func retryPending() async {
        let pending = pendingMessages.filter { $0.status == .pending || $0.status == .failed }
        for message in pending {
            markSending(id: message.id)
            // Notify ChatViewModel to actually send
            NotificationCenter.default.post(
                name: .pendingChatReadyToSend,
                object: nil,
                userInfo: ["message": message]
            )
        }
    }
    
    // MARK: - CoreData Helpers
    
    private func updateEntity(id: UUID, status: String) {
        let request: NSFetchRequest<PendingChatMessage> = PendingChatMessage.fetchRequest()
        request.predicate = NSPredicate(format: "id == %@", id as CVarArg)
        if let entity = try? context.fetch(request).first {
            entity.status = status
            try? context.save()
        }
    }
    
    private func deleteEntity(id: UUID) {
        let request: NSFetchRequest<PendingChatMessage> = PendingChatMessage.fetchRequest()
        request.predicate = NSPredicate(format: "id == %@", id as CVarArg)
        if let entity = try? context.fetch(request).first {
            context.delete(entity)
            try? context.save()
        }
    }
}

extension Notification.Name {
    static let pendingChatReadyToSend = Notification.Name("pendingChatReadyToSend")
}
```

func sendMessage(_ text: String) async {
    let trimmed = text.trimmingCharacters(in: .whitespacesAndNewlines)
    guard !trimmed.isEmpty else { return }
    
    // Check if offline — queue instead of send (D-31)
    if !NetworkMonitor.shared.isConnected {
        PendingChatQueue.shared.enqueue(text: trimmed, conversationID: currentConversationID)
        // Add local user bubble immediately
        addUserMessage(trimmed, isPending: true)
        return
    }
    
    // Existing send logic...
    addUserMessage(trimmed, isPending: false)
    // ... AI service call ...
}

Add listener for retry notifications:
```swift
private func listenForRetries() {
    NotificationCenter.default.addObserver(
        forName: .pendingChatReadyToSend,
        object: nil,
        queue: .main
    ) { [weak self] notification in
        guard let message = notification.userInfo?["message"] as? PendingMessage else { return }
        Task { @MainActor in
            await self?.retrySendMessage(message)
        }
    }
}

private func retrySendMessage(_ pending: PendingMessage) async {
    do {
        // Actual AI send
        let response = try await aiService.sendChatMessage(pending.text)
        PendingChatQueue.shared.remove(id: pending.id)
        // Update UI: remove pending indicator, add AI response
        updatePendingBubble(id: pending.id, success: true)
        addAIResponse(response)
    } catch {
        PendingChatQueue.shared.markFailed(id: pending.id)
        updatePendingBubble(id: pending.id, success: false)
    }
}
```

// In the messages list, after regular messages:
ForEach(PendingChatQueue.shared.pendingMessages) { pending in
    VStack(alignment: .trailing, spacing: 4) {
        // User bubble (right-aligned, same style as regular user message)
        Text(pending.text)
            .padding(12)
            .background(Color(red: 0.3, green: 0.69, blue: 0.31).opacity(0.8))
            .foregroundColor(.white)
            .cornerRadius(16)
        
        // Status indicator
        HStack(spacing: 4) {

            case .pending:
                Image(systemName: "clock").font(.caption2)

    }

    .padding(.horizontal)
